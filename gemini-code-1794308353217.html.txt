  <script>
    let peer = null, conn = null;
    let myName = "", myRole = "host";
    let isReady = false, isFinished = false;
    let finishedCount = 0, totalPlayers = 2;
    let currentStep = 1, timerVal = 90, timerInterval = null;

    let canvas, ctx, drawing = false, currentColor = 'black';
    let strokeHistory = [];

    let audioCtx = null, isBgmPlaying = false, bgmTimer = null;

    function initAudio() {
      if(!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    }

    function playSynthNote(freq, type, duration, time) {
      if(!audioCtx) return;
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = type;
      osc.frequency.setValueAtTime(freq, time);
      gain.gain.setValueAtTime(0.06, time);
      gain.gain.exponentialRampToValueAtTime(0.001, time + duration);
      osc.connect(gain); gain.connect(audioCtx.destination);
      osc.start(time); osc.stop(time + duration);
    }

    function startArcadeBGM() {
      initAudio();
      if(audioCtx.state === 'suspended') audioCtx.resume();
      const notes = [261.63, 329.63, 392.00, 523.25, 440.00, 349.23, 392.00, 293.66];
      let step = 0;
      function loop() {
        if(!isBgmPlaying) return;
        const now = audioCtx.currentTime;
        playSynthNote(notes[step % notes.length], 'square', 0.12, now);
        if(step % 2 === 0) playSynthNote(130.81, 'sawtooth', 0.18, now);
        step++; bgmTimer = setTimeout(loop, 180);
      }
      isBgmPlaying = true; loop();
    }

    function toggleBGM() {
      if(isBgmPlaying) {
        isBgmPlaying = false; clearTimeout(bgmTimer);
        document.getElementById('bgmBtn').innerText = '🎵 BGM';
      } else {
        startArcadeBGM();
        document.getElementById('bgmBtn').innerText = '🔊 BGM';
      }
    }

    function createRoom() {
      myName = document.getElementById('nickname').value.trim();
      if(!myName) return alert('닉네임을 입력하세요!');
      myRole = "host";

      const code = Math.random().toString(36).substring(2, 6).toUpperCase();
      peer = new Peer("GARTIC-" + code);

      peer.on('open', () => {
        document.getElementById('screenLobby').style.display = 'none';
        document.getElementById('screenRoom').style.display = 'flex';
        document.getElementById('roomCodeText').innerText = code;
        document.getElementById('qrContainer').innerHTML = "";
        new QRCode(document.getElementById('qrContainer'), {
          text: window.location.href.split('?')[0] + "?room=" + code,
          width: 130, height: 130
        });
      });

      peer.on('connection', (c) => {
        conn = c;
        conn.on('data', handleData);
      });
    }

    function joinRoom() {
      myName = document.getElementById('nickname').value.trim();
      const code = document.getElementById('roomInput').value.trim().toUpperCase();
      if(!myName || !code) return alert('닉네임과 방 코드를 입력하세요!');
      myRole = "guest";

      peer = new Peer();
      peer.on('open', () => {
        conn = peer.connect("GARTIC-" + code);
        conn.on('open', () => {
          document.getElementById('screenLobby').style.display = 'none';
          document.getElementById('screenRoom').style.display = 'flex';
          document.getElementById('roomCodeText').innerText = code;
          conn.on('data', handleData);
        });
      });
    }

    function toggleReady() {
      isReady = !isReady;
      const btn = document.getElementById('btnReady');
      btn.innerText = isReady ? "준비 취소" : "준비 완료";
      btn.style.background = isReady ? "#ff4757" : "var(--kakao-yellow)";
      btn.style.color = isReady ? "#fff" : "var(--kakao-brown)";
      
      if(conn) conn.send({ type: 'READY_CHANGE', ready: isReady });
      checkAllReady();
    }

    function checkAllReady() {
      if(isReady) startGame();
    }

    function handleData(data) {
      if(data.type === 'SYNC_FINISH') {
        finishedCount = data.count;
        updateBadge();
        if(finishedCount >= totalPlayers) nextPhaseAll();
      }
    }

    function updateBadge() {
      const badge = document.getElementById('completionBadge');
      badge.style.display = 'inline-block';
      badge.innerText = `완료 ${finishedCount}/${totalPlayers}명`;
    }

    function startGame() {
      document.getElementById('screenRoom').style.display = 'none';
      document.getElementById('screenGame').style.display = 'flex';
      if(!isBgmPlaying) toggleBGM();
      startTimer();
      initCanvas();
      updateBadge();
    }

    function startTimer() {
      timerVal = 90;
      document.getElementById('timerText').innerText = timerVal;
      clearInterval(timerInterval);
      timerInterval = setInterval(() => {
        timerVal--;
        document.getElementById('timerText').innerText = timerVal;
        if(timerVal <= 0) {
          clearInterval(timerInterval);
          if(!isFinished) submitTurn();
        }
      }, 1000);
    }

    function submitTurn() {
      isFinished = true;
      finishedCount++;
      updateBadge();

      document.getElementById('btnSubmit').style.display = 'none';
      document.getElementById('btnEdit').style.display = 'block';

      if(conn) conn.send({ type: 'SYNC_FINISH', count: finishedCount });
      if(finishedCount >= totalPlayers) nextPhaseAll();
    }

    function editTurn() {
      isFinished = false;
      finishedCount = Math.max(0, finishedCount - 1);
      updateBadge();

      document.getElementById('btnSubmit').style.display = 'block';
      document.getElementById('btnEdit').style.display = 'none';

      if(conn) conn.send({ type: 'SYNC_FINISH', count: finishedCount });
    }

    function nextPhaseAll() {
      clearInterval(timerInterval);
      finishedCount = 0;
      isFinished = false;
      document.getElementById('btnSubmit').style.display = 'block';
      document.getElementById('btnEdit').style.display = 'none';

      if(currentStep === 1) {
        currentStep = 2;
        document.getElementById('phaseTitle').innerText = "제시어를 보고 그림을 그려주세요!";
        document.getElementById('promptArea').style.display = 'none';
        document.getElementById('drawArea').style.display = 'block';
        startTimer();
        updateBadge();
      } else {
        document.getElementById('screenGame').style.display = 'none';
        document.getElementById('screenStory').style.display = 'flex';
        showStoryResult();
      }
    }

    function initCanvas() {
      canvas = document.getElementById('canvas');
      ctx = canvas.getContext('2d');
      ctx.lineWidth = 4; ctx.lineCap = 'round';

      function getPos(e) {
        const rect = canvas.getBoundingClientRect();
        const clientX = e.touches ? e.touches[0].clientX : e.clientX;
        const clientY = e.touches ? e.touches[0].clientY : e.clientY;
        return { x: clientX - rect.left, y: clientY - rect.top };
      }

      function start(e) {
        if(isFinished) return;
        drawing = true;
        const pos = getPos(e);
        ctx.beginPath(); ctx.moveTo(pos.x, pos.y);
        strokeHistory.push({ x: pos.x, y: pos.y, color: currentColor, isNewStroke: true });
      }

      function move(e) {
        if(!drawing || isFinished) return;
        const pos = getPos(e);
        ctx.lineTo(pos.x, pos.y);
        ctx.strokeStyle = currentColor; ctx.stroke();
        strokeHistory.push({ x: pos.x, y: pos.y, color: currentColor, isNewStroke: false });
      }

      function stop() { drawing = false; }

      canvas.addEventListener('mousedown', start);
      canvas.addEventListener('mousemove', move);
      canvas.addEventListener('mouseup', stop);
      canvas.addEventListener('touchstart', (e) => { e.preventDefault(); start(e); });
      canvas.addEventListener('touchmove', (e) => { e.preventDefault(); move(e); });
      canvas.addEventListener('touchend', stop);
    }

    function setColor(color, el) {
      currentColor = color;
      document.querySelectorAll('.color-btn').forEach(b => b.classList.remove('active'));
      el.classList.add('active');
    }

    function clearCanvas() {
      if(isFinished) return;
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      strokeHistory = [];
    }

    function showStoryResult() {
      const promptVal = document.getElementById('inputPrompt').value.trim() || '우주를 나는 고양이';

      document.getElementById('name1').innerText = myName;
      document.getElementById('avatar1').innerText = myName.charAt(0);
      document.getElementById('textTopic').innerText = `"${promptVal}"`;

      document.getElementById('name2').innerText = "친구 (그림)";
      document.getElementById('avatar2').innerText = "친";

      document.getElementById('name3').innerText = "친구 (정답)";
      document.getElementById('avatar3').innerText = "친";
      document.getElementById('textGuess').innerText = '"날개 달린 야옹이"';

      replayDrawing5x();
    }

    /* 5배속 타임랩스 렌더링 */
    function replayDrawing5x() {
      const repCvs = document.getElementById('replayCanvas');
      const repCtx = repCvs.getContext('2d');
      repCtx.clearRect(0, 0, repCvs.width, repCvs.height);
      repCtx.lineWidth = 4; repCtx.lineCap = 'round';

      if(strokeHistory.length === 0) {
        repCtx.strokeStyle = '#ff4757';
        repCtx.beginPath(); repCtx.arc(130, 95, 45, 0, Math.PI * 2); repCtx.stroke();
        return;
      }

      let idx = 0;
      const speed = 5; // 5배속
      const interval = setInterval(() => {
        for(let i = 0; i < speed; i++) {
          if(idx >= strokeHistory.length) {
            clearInterval(interval);
            break;
          }
          const pt = strokeHistory[idx];
          if(pt.isNewStroke) {
            repCtx.beginPath(); repCtx.moveTo(pt.x, pt.y);
          } else {
            repCtx.strokeStyle = pt.color;
            repCtx.lineTo(pt.x, pt.y); repCtx.stroke();
          }
          idx++;
        }
      }, 16);
    }

    function resetToRoom() {
      currentStep = 1; strokeHistory = []; isReady = false; isFinished = false;
      document.getElementById('btnReady').innerText = "준비 완료";
      document.getElementById('btnReady').style.background = "var(--kakao-yellow)";
      document.getElementById('btnReady').style.color = "var(--kakao-brown)";
      document.getElementById('screenStory').style.display = 'none';
      document.getElementById('screenRoom').style.display = 'flex';
      document.getElementById('promptArea').style.display = 'block';
      document.getElementById('drawArea').style.display = 'none';
    }
